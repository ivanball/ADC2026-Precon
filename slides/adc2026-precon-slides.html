The whole Result type (~40 lines)

public sealed record Error(string Code, string Message)
{
    public static readonly Error None = new(string.Empty, string.Empty);
}

public class Result
{
    protected Result(bool isSuccess, Error error) { /* guards: success has None, failure has not */ }
    public bool IsSuccess { get; }
    public bool IsFailure => !IsSuccess;
    public Error Error { get; }
    public static Result Success() ...                      public static Result Failure(Error e) ...
    public static Result<T> Success<T>(T value) ...      public static Result<T> Failure<T>(Error e) ...
}

public class Result<TValue> : Result
{
    public TValue Value => IsSuccess ? _value!
        : throw new InvalidOperationException("Accessed Value on a failed result.");  // a caller BUG throws
}

    ICommandHandler<TCommand, TResponse> inner,
    IEnumerable<IValidator<TCommand>> validators)
    : ICommandHandler<TCommand, TResponse>          // <-- same interface
    where TCommand : ICommand<TResponse>

            : inner.Handle(command, ct);
